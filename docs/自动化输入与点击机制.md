# 自动化输入与点击机制总结

本文档总结 Chrome 扩展中如何实现自动化输入框填写和按钮点击。

---

## 一、自动化输入框填写

### 1. 核心函数：`fillInput()`

位置：`content/utils.js`

```js
function fillInput(el, value) {
  // 1. 获取 HTMLInputElement 原生的 value setter
  const nativeInputValueSetter = Object.getOwnPropertyDescriptor(
    window.HTMLInputElement.prototype,
    'value'
  ).set;

  // 2. 调用原生 setter，绕过 React 等框架的拦截
  nativeInputValueSetter.call(el, value);

  // 3. 手动触发 input 和 change 事件（冒泡），让框架感知变化
  el.dispatchEvent(new Event('input', { bubbles: true }));
  el.dispatchEvent(new Event('change', { bubbles: true }));
}
```

**为什么不能直接 `el.value = xxx`？**

React 等现代前端框架会拦截 input 的 value setter，直接赋值不会触发框架状态更新。通过 `Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set` 获取浏览器原生的 setter，可以绕过框架拦截，再手动 dispatch 事件确保框架同步感知。

#### 2.5 React Aria 生日选择器（年/月/日三个下拉）

位置：`content/signup-page.js`

```js
async function setReactAriaBirthdaySelect(control, value) {
  const desiredValue = String(value);
  const option = Array.from(control.nativeSelect.options)
    .find((item) => item.value === desiredValue);

  control.nativeSelect.value = desiredValue;
  option.selected = true;
  control.nativeSelect.dispatchEvent(new Event('input', { bubbles: true }));
  control.nativeSelect.dispatchEvent(new Event('change', { bubbles: true }));
}
```

通过 `.react-aria-Select` 类名定位根元素，找到其中的 `hidden-select-container` 内的原生 `<select>` 来写入。

#### 2.6 SpinButton 数字输入器（生日的另一种 UI）

位置：`content/signup-page.js`

```js
async function setSpinButton(el, value) {
  el.focus();
  document.execCommand('selectAll', false, null);

  for (const char of String(value)) {
    el.dispatchEvent(new KeyboardEvent('keydown', { key: char, bubbles: true }));
    el.dispatchEvent(new InputEvent('beforeinput', { inputType: 'insertText', data: char, bubbles: true }));
    el.dispatchEvent(new InputEvent('input', { inputType: 'insertText', data: char, bubbles: true }));
    await sleep(50);
  }

  el.dispatchEvent(new KeyboardEvent('keyup', { key: 'Tab', code: 'Tab', bubbles: true }));
  el.blur();
}
```

模拟键盘逐字符输入，适用于 `[role="spinbutton"]` 类型的数字控件。

## 二、元素查找策略

### 1. 等待元素出现：`waitForElement()`

位置：`content/utils.js`

```js
function waitForElement(selector, timeout = 10000) {
  // 1. 先检查元素是否已存在
  const existing = document.querySelector(selector);
  if (existing) return existing;

  // 2. 使用 MutationObserver 监听 DOM 变化
  const observer = new MutationObserver(() => {
    const el = document.querySelector(selector);
    if (el) { observer.disconnect(); resolve(el); }
  });
  observer.observe(document.body, { childList: true, subtree: true });

  // 3. 超时处理
  const timer = setTimeout(() => {
    observer.disconnect();
    reject(new Error(`等待 ${selector} 超时`));
  }, timeout);
}
```

### 2. 按文本查找：`waitForElementByText()`

位置：`content/utils.js`

```js
function waitForElementByText(containerSelector, textPattern, timeout = 10000) {
  function search() {
    const candidates = document.querySelectorAll(containerSelector);
    for (const el of candidates) {
      if (textPattern.test(el.textContent)) return el;
    }
    return null;
  }

  const existing = search();
  if (existing) return existing;

  // MutationObserver 监听 + 轮询 search()
}
```

### 3. 可见性检查

位置：`content/signup-page.js`

```js
function isVisibleElement(el) {
  const style = window.getComputedStyle(el);
  const rect = el.getBoundingClientRect();
  return style.display !== 'none'
    && style.visibility !== 'hidden'
    && rect.width > 0 && rect.height > 0;
}
```

### 4. 按钮文本提取

位置：`content/signup-page.js`

```js
function getActionText(el) {
  return [
    el?.textContent,
    el?.value,
    el?.getAttribute?.('aria-label'),
    el?.getAttribute?.('title'),
  ].filter(Boolean).join(' ').trim();
}
```

从多个属性中提取文本：textContent、value、aria-label、title。

### 5. 按钮启用状态检查

```js
function isActionEnabled(el) {
  return Boolean(el) && !el.disabled && el.getAttribute('aria-disabled') !== 'true';
}
```

## 三、自动化按钮点击

### 1. 核心函数：`simulateClick()`

位置：`content/utils.js`

```js
function simulateClick(el) {
  const form = el.form || el.closest?.('form') || null;

  // 获取激活策略（根据元素类型和上下文决定点击方式）
  const strategy = typeof getActivationStrategy === 'function'
    ? getActivationStrategy({
        tagName: el.tagName,
        type: el.getAttribute?.('type') || '',
        hasForm: Boolean(form),
        pathname: location.pathname || '',
      })
    : { method: 'click' };

  let method = strategy.method || 'click';

  // 方式 A：form.requestSubmit() — 触发完整表单提交流程
  if (method === 'requestSubmit' && form && typeof form.requestSubmit === 'function') {
    form.requestSubmit(el);
  }
  // 方式 B：原生 click()
  else if (typeof el.click === 'function') {
    el.click();
  }
  // 方式 C：dispatchEvent 模拟 MouseEvent
  else {
    el.dispatchEvent(new MouseEvent('click', { bubbles: true, cancelable: true }));
  }
}
```

三种点击方式按优先级：
1. **`form.requestSubmit(el)`** — 触发完整的 HTML 表单提交（包括验证）
2. **`el.click()`** — 原生 DOM 点击
3. **`dispatchEvent(new MouseEvent('click'))`** — 模拟鼠标点击事件

### 2. Chrome Debugger 坐标点击（Step 8 专用）

位置：`background.js`

对于 OAuth 同意页的"继续"按钮，使用 Chrome DevTools Protocol 模拟真实鼠标点击：

```js
async function clickWithDebugger(tabId, rect) {
  const target = { tabId };

  // 1. 附加调试器
  await chrome.debugger.attach(target, '1.3');

  try {
    const x = Math.round(rect.centerX);
    const y = Math.round(rect.centerY);

    // 2. 将标签页置于前台
    await chrome.debugger.sendCommand(target, 'Page.bringToFront');

    // 3. 移动鼠标到目标位置
    await debugger.sendCommand(target, 'Input.dispatchMouseEvent', {
      type: 'mouseMoved', x, y, button: 'none', buttons: 0, clickCount: 0,
    });

    // 4. 按下鼠标左键
    await debugger.sendCommand(target, 'Input.dispatchMouseEvent', {
      type: 'mousePressed', x, y, button: 'left', buttons: 1, clickCount: 1,
    });

    // 5. 释放鼠标左键
    await debugger.sendCommand(target, 'Input.dispatchMouseEvent', {
      type: 'mouseReleased', x, y, button: 'left', buttons: 0, clickCount: 1,
    });
  } finally {
    await chrome.debugger.detach(target);
  }
}
```

### 3. Step 8 完整点击链路

Step 8 使用多策略循环尝试：

```js
const STEP8_STRATEGIES = [
  { mode: 'content', strategy: 'requestSubmit', label: 'form.requestSubmit' },
  { mode: 'debugger', label: 'debugger click' },
  { mode: 'content', strategy: 'nativeClick', label: 'element.click' },
  { mode: 'content', strategy: 'dispatchClick', label: 'dispatch click' },
  { mode: 'debugger', label: 'debugger click retry' },
];
```

完整流程：

1. **Background** → 循环尝试不同策略
2. **Background** → 发送 `STEP8_FIND_AND_CLICK` 消息到 content script
3. **Content Script** → 查找按钮（`findContinueButton()`），等待可见、等待启用
4. **Content Script** → 滚动到视口、聚焦、等待位置稳定
5. **Content Script** → 计算坐标（`getSerializableRect()`），返回 `{ centerX, centerY, width, height }`
6. **Background** → 使用 CDP `Input.dispatchMouseEvent` 执行真实点击
7. **Background** → 监听 `chrome.webNavigation.onBeforeNavigate` 捕获回调

## 四、Background 与 Content Script 的协作

### 1. 消息协议

| 方向 | 消息类型 | 用途 |
|------|----------|------|
| Background → Content | `EXECUTE_STEP` | 触发执行某个步骤（填写+点击） |
| Background → Content | `FILL_CODE` | 填写验证码 |
| Background → Content | `STEP8_FIND_AND_CLICK` | 查找按钮并返回坐标 |
| Content → Background | `STEP_COMPLETE` | 步骤完成信号 |
| Content → Background | `STEP_ERROR` | 步骤错误信号 |
| Content → Background | `CONTENT_SCRIPT_READY` | 内容脚本就绪通知 |

### 2. 弹性发送：`sendToContentScriptResilient()`

位置：`background/tab-runtime.js`

```js
async function sendToContentScriptResilient(source, message, options = {}) {
  const { timeoutMs = 30000, retryDelayMs = 600 } = options;
  const start = Date.now();

  while (Date.now() - start < timeoutMs) {
    try {
      return await sendToContentScript(source, message);
    } catch (err) {
      // 可重试的传输错误（如页面跳转、port 关闭等）
      if (!isRetryableContentScriptTransportError(err)) throw err;
      await new Promise((r) => setTimeout(r, retryDelayMs));
    }
  }
  throw new Error('等待内容脚本重新就绪超时');
}
```

### 3. 命令排队机制

当 content script 尚未就绪时，消息会进入队列等待：

```js
// 入队
function queueCommand(source, message, timeout = 15000) {
  pendingCommands.set(source, { message, resolve, reject, timer });
}

// content script 就绪时刷新
function flushCommand(source, tabId) {
  const pending = pendingCommands.get(source);
  if (pending) {
    sendTabMessageWithTimeout(tabId, source, pending.message);
  }
}
```

## 五、完整流程示例：Step 2 填写邮箱并提交

```
1. Background 解析邮箱地址
2. 确保 signup-page 标签存在且 content script 已注入
3. 发送 EXECUTE_STEP 消息到 content script
4. Content script 中：
   a. waitForElement() 等待邮箱输入框出现
   b. fillInput(emailInput, email) 填写邮箱
   c. 查找"继续"按钮（文本匹配 continue/继续）

```

## 六、关键技术要点总结

| 技术点 | 解决方案 |
|--------|----------|
| 绕过 React 输入拦截 | 使用 `Object.getOwnPropertyDescriptor` 获取原生 setter |
| 确保框架感知输入 | 手动 dispatch `input` + `change` 事件 |
| 等待动态渲染元素 | `MutationObserver` 监听 DOM 变化 |
| 查找文本匹配的按钮 | 遍历候选元素，用正则匹配 `getActionText()` |
| 确保按钮可点击 | 检查 `isVisibleElement()` + `isActionEnabled()` |
| 表单提交 | 优先 `form.requestSubmit()`，备选 `el.click()` |
| 跨域/沙盒页面的点击 | Chrome Debugger CDP `Input.dispatchMouseEvent` |
| Content script 未就绪 | 命令排队 + 弹性重试 |
| 页面跳转中断 | 可重试传输错误识别 + 自动恢复 |
